<script>
$('#buscarTicket').on('keyup', function() {
    var q = $(this).val().trim();
    if (q.length < 2) {
        $('#resultadoTicket').hide().empty();
        return;
    }
    $.getJSON('/api/buscar_tickets_json', { q: q }, function(data) {
        var html = '';
        data.forEach(function(t) {
            html += '<a href="#" class="list-group-item list-group-item-action" data-id="' + t.id + '">';
            html += '<strong>' + t.num_ticket + '</strong> - ' + t.cliente + ' (' + t.motivo_ticket + ')';
            html += '</a>';
        });
        if (html) {
            $('#resultadoTicket').html(html).show();
        } else {
            $('#resultadoTicket').hide().empty();
        }
    });
});

$(document).on('click', '#resultadoTicket a', function(e) {
    e.preventDefault();
    var id = $(this).data('id');
    var texto = $(this).text();
    $('#ticket_id').val(id);
    $('#buscarTicket').val(texto);
    $('#resultadoTicket').hide();
});
</script>
